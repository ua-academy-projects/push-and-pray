# Controller-side Python requirements for the OilScope Ansible tooling.
#
# The google.cloud.gcp_compute inventory plugin used by
# inventory/oilscope.yml talks to the Compute Engine API through these
# libraries, authenticating with Application Default Credentials:
#
#     pip install -r infrastructure/ansible/requirements.txt
#     gcloud auth application-default login
#
google-auth>=2.0.0
requests>=2.25.0

# kubernetes.core's k8s modules talk to the API through this library. The helm
# and helm_info modules shell out to the helm binary instead, so they need
# helm on PATH rather than anything here - see docs/k3s-deployment.md for the
# local tool versions this deployment was verified against.
kubernetes>=31.0.0
PyYAML>=6.0

# The amazon.aws.aws_ec2 inventory plugin used by inventory/oilscope-aws.yml
# talks to the EC2 API through these libraries, authenticating through the
# standard boto3 credential chain (environment variables, a shared
# credentials/config file, or an assumed/instance role):
#
#     pip install -r infrastructure/ansible/requirements.txt
#
# Floors match what amazon.aws itself requires at the version pinned in
# requirements.yml (checked against its own requirements.txt) - installing
# collections does not upgrade an already-installed, already-satisfying-on-
# paper but actually-too-old SDK, so a lower floor here would silently work
# until it doesn't.
boto3>=1.35.0
botocore>=1.35.0

# The azure.azcollection.azure_rm inventory plugin used by
# inventory/oilscope-azure.yml is not pinned here, and deliberately so: that
# collection's module_utils imports its entire SDK surface inside one try
# block, so a partial install leaves the plugin unusable with a message that
# names none of the packages actually missing. Install the collection's own
# requirements file instead, after installing the collection:
#
#     ansible-galaxy collection install -r infrastructure/ansible/requirements.yml
#     pip install -r ~/.ansible/collections/ansible_collections/azure/azcollection/requirements.txt
#
# (adjust the path if collections are installed somewhere other than the
# default; `ansible-galaxy collection list azure.azcollection` prints it.)
#
# Azure credentials come from ARM_* environment variables or `az login`; the
# Azure CLI is also what the controller-side secret upload and the managed
# migration's administrator read use.
